      싱글톤(Singleton) 이란?

      "이 세상에 나(Instance)는 하나만 존재한다."

      싱글톤(Singleton) 패턴은 클래스의 인스턴스를 오직 하나만 생성하고, 어디서든 그 인스턴스에 접근할 수 있도록 보장하는 패턴입니다.

      주로 게임 매니저(GameManager), 오디오 매니저, 네트워크 연결 등 시스템 전반에서 공유되어야 하는 유일한 객체를 만들 때 사용됩니다.

      싱글톤 구현 예시 (Unity/C#)

      1. 기본적인 C# 싱글톤

      DataManager.cs
      public class DataManager
{
    // 1. 외부에서 직접 생성하지 못하도록 생성자를 private으로 설정
    private DataManager() { }

    // 2. 유일한 인스턴스를 저장할 static 변수
    private static DataManager _instance;

    // 3. 외부에서 접근할 수 있는 프로퍼티
    public static DataManager Instance
    {
        get
        {
            if (_instance == null)
            {
                _instance = new DataManager();
            }
            return _instance;
        }
    }

    public void SaveGame() { /* ... */ }
}

      C#에서 싱글턴을 사용 할 때 기초적인 형태로서, get시에 필요할 때만 생성하는 지연 초기화(Lazy Initialization) 방식을 사용합니다.

      2. Unity MonoBehaviour 싱글톤

      GameManager.cs
      using UnityEngine;

public class GameManager : MonoBehaviour
{
    private static GameManager _instance;

    public static GameManager Instance
    {
        get
        {
            if (_instance == null)
            {
                // 1. 씬에 이미 배치된 매니저가 있는지 찾아본다.
                _instance = FindObjectOfType<GameManager>();

                // 2. 찾아도 없다면 그때 새로 만든다.
                if (_instance == null)
                {
                    GameObject go = new GameObject("GameManager");
                    _instance = go.AddComponent<GameManager>();
                }
            }
            return _instance;
        }
    }

    private void Awake()
    {
        // 중복 생성 방지
        if (_instance != null && _instance != this)
        {
            Destroy(gameObject);
            return;
        }
        
        _instance = this;
        DontDestroyOnLoad(gameObject); // 씬 전환 시 파괴 방지
    }
}

      C#에서 싱글턴을 사용 할 때 기초적인 형태로서, 씬에 미리 배치해둔 오브젝트가 있다면 그것을 사용하고, 없다면 새로 만듭니다.

      3. 제네릭 모노 싱글톤 (실전 완성형)

      부모 클래스 (MonoSingleton)

      MonoSingleton.cs
      using UnityEngine;

// T는 MonoBehaviour를 상속받은 클래스여야 한다
public class MonoSingleton<T> : MonoBehaviour where T : MonoBehaviour
{
    // 스레드 안전성을 위한 잠금 객체
    private static readonly object _lock = new object();
    private static T _instance = null;

    // 종료 시 재생성 방지 플래그
    private static bool _applicationIsQuitting = false;

    [SerializeField]
    [Tooltip("체크하면 씬이 변경되어도 파괴되지 않습니다.")]
    protected bool isDontDestroy = false;

    public static T Instance
    {
        get
        {
            // 종료 과정 중에 호출되면 null 반환 (에디터 종료 시 에러 방지)
            if (_applicationIsQuitting)
            {
                Debug.LogWarning($"[MonoSingleton] '{typeof(T)}' 인스턴스는 이미 종료 중에 파괴되었습니다. null을 반환합니다.");
                return null;
            }

            // 스레드 안전성 확보 (Thread-Safe)
            lock (_lock)
            {
                if (_instance == null)
                {
                    // 1. 씬에 있는 오브젝트 검색
                    _instance = FindObjectOfType<T>();

                    // 2. 없으면 새로 생성
                    if (_instance == null)
                    {
                        GameObject singletonObject = new GameObject();
                        _instance = singletonObject.AddComponent<T>();
                        singletonObject.name = $"{typeof(T).Name} (Singleton)";
                    }
                }
                return _instance;
            }
        }
    }

    protected virtual void Awake()
    {
        if (_instance == null)
        {
            // 내가 첫 번째 인스턴스라면 등록
            _instance = this as T;
            if (isDontDestroy)
            {
                DontDestroyOnLoad(gameObject);
            }
        }
        else if (_instance != this)
        {
            // 이미 다른 인스턴스가 있다면 나는 파괴 (중복 방지)
            Debug.LogWarning($"[MonoSingleton] '{GetType().Name}' 인스턴스가 이미 존재하여 중복된 객체를 파괴합니다.");
            Destroy(gameObject);
        }
    }

    protected virtual void OnDestroy()
    {
        if (_instance == this)
        {
            _instance = null;
        }
    }

    /// <summary>
    /// 앱 종료 시 플래그 설정하여 재생성 방지
    /// </summary>
    protected virtual void OnApplicationQuit()
    {
        _applicationIsQuitting = true;
    }
}

      사용 예시 (자식 클래스)

      SoundManager.cs
      // 이제 : MonoBehaviour 대신 : MonoSingleton<SoundManager>를 상속받습니다.
public class SoundManager : MonoSingleton<SoundManager>
{
    // 필요하다면 Inspector에서 Is Dont Destroy를 체크할 수 있습니다.
    
    public void PlayMusic()
    {
        Debug.Log("음악 재생!");
    }
    
    // Awake 재정의 시 base.Awake() 필수!
    protected override void Awake()
    {
        base.Awake(); 
        // 추가 초기화 로직...
    }
}

      MonoSingleton이라는 제네릭 클래스를 만들어서 사용 할 때 부모로 호출하면 바로 싱글턴 패턴이 적용 될 수 있게 만들었습니다.

      실무에서 사용하는 가장 강력하고 안전한 버전입니다.

      • 제네릭(Generic): 상속만 받으면 바로 사용 가능
      • 스레드 안전(Thread-Safe): lock을 사용하여 멀티스레드 환경에서도 안전
      • 종료 처리(Quit Handling): 앱 종료 시 인스턴스가 재생성되는 ‘유령 객체(Ghost Object)’ 문제 방지
      • 옵션 제공: isDontDestroy 변수로 파괴 여부 선택 가능

      장점과 단점

      장점

      1. 전역 접근: ClassName.Instance를 통해 어디서든 쉽게 접근할 수 있습니다.
      2. 데이터 공유: 하나의 인스턴스를 공유하므로 데이터 동기화가 쉽습니다.
      3. 메모리 절약: 고정된 메모리 영역을 사용하여 객체를 한 번만 생성하므로 불필요한 메모리 낭비를 막을 수 있습니다.

      단점

      1. 높은 결합도 (Tight Coupling): 싱글톤을 사용하는 클래스는 싱글톤 객체에 강하게 의존하게 됩니다. 이는 코드 수정과 유지보수를 어렵게 만듭니다.
      2. 테스트의 어려움: 전역 상태를 가지므로 단위 테스트(Unit Test)를 진행할 때 각 테스트가 서로 영향을 줄 수 있어 모킹(Mocking)이나 격리가 어렵습니다.
      3. 숨겨진 의존성: 생성자나 인자로 전달받지 않고 내부에서 Instance를 호출해 사용하므로, 코드를 까보지 않으면 어떤 의존성이 있는지 알기 어렵습니다.

      전체 요약

      1. 하나의 인스턴스만 필요 할 때 사용하면 편리
      2. 막 쓰면 스파게티 코드의 주범, 꼭 필요한 곳(매니저급)에만 사용
