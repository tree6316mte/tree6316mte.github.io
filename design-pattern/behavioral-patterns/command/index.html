      커맨드

      커맨드(Command) 패턴이란?

      스타크래프트에서는 Shift를 사용해서 명령을 대기시켜 놓을 수 있다

      커맨드(Command) 패턴은 실행 가능한 함수(명령) 를 저장가능한 객체로 만드는 것을 중점으로 둔 디자인 패턴입니다.

      쉽게 말해, 명령을 한번 쓰고 버리는 것이 아닌 저장할 수 있는 곳에 저장해 두는 것입니다.

      함수가 다른 변수나 값처럼 된다면?

      • 저장: 큐, 리스트, 스택에 담아둘 수 있음
      • 지연 실행: 지금 당장이 아니라 나중에 실행 가능
      • 전달: 다른 메서드의 매개변수로 넘길 수 있음
      • 취소/되돌리기: 객체니까 “반대 동작"도 같이 들고 다닐 수 있음 (Undo)
      • 기록/재생: 실행 이력을 객체 리스트로 남겨서 재생(replay) 가능

      구성 요소

      이 패턴은 크게 5가지 핵심 요소로 구성된다

      1. Command (인터페이스): 모든 명령 객체가 구현해야 하는 기본 인터페이스로, 주로 Execute()와 Undo() 메서드를 정의
      2. ConcreteCommand (구체적 명령): Command 인터페이스를 구현하며, 수신자(Receiver)와 그 수신자가 수행해야 할 구체적인 동작을 연결하는 역할
      3. Receiver (수신자): 실제 비즈니스 로직을 수행하고 명령의 요구 사항을 처리하는 객체 (예: Unit 클래스)
      4. Invoker (발동자): 명령을 저장하고 실행을 요청하는 객체로, 어떤 명령을 언제 실행할지 관리 (예: GameLogic 클래스)
      5. Client (클라이언트): 구체적인 명령 객체를 생성하고 발동자와 수신자를 연결하는 초기화 역할을 담당

      커맨드 패턴 적용 예시

      유닛이 움직이고 건물을 지을 수 있는 상황을 가정해 보겠습니다.

      커맨드 패턴 없는 코드

      GameLogic.cs
      using UnityEngine;
using UnityEngine.InputSystem;

public class Unit
{
    public Vector2 position = new();

    // 명령어 목록
    public void MoveTo(Vector2 position)
    {
        this.position = position;
        Debug.Log($"(X:{position.x},Y:{position.y})로 움직임!");
    }
    public void Build(Vector2 pos, string name)
    {
        Debug.Log($"[Unit] {pos} 위치에 '{name}'을(를) 건설했습니다.");
    }
    public void RemoveBuilding(Vector2 pos, string name)
    {
        Debug.Log($"[Unit] {pos} 위치의 '{name}'을(를) 삭제했습니다.");
    }
}

public class GameLogic : MonoBehaviour
{
    private Unit workman = new();

    void Update()
    {
        // Z 키를 누르면 오른쪽으로 1칸 이동
        if (Keyboard.current.zKey.wasPressedThisFrame)
        {
            Vector2 targetPos = workman.position + new Vector2(1, 0);
            workman.MoveTo(targetPos);
        }

        // X 키를 누르면 현재 위치에 배럭 건설
        if (Keyboard.current.xKey.wasPressedThisFrame)
        {
            Vector2 targetPos = workman.position;
            workman.Build(targetPos, "배럭");
        }
    }
}

      위 코드는 일꾼을 Z키로 움직이고 X키로 건물을 건설 할 수 있게 만든 코드입니다.

      하지만 과거로 돌아가려고 하니 이전에 무엇을 했는지 담을 수 있는 방법이 없습니다.

      커맨드 패턴 코드

      GameLogic.cs
      using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;

// 커맨드 인터페이스
public interface ICommand
{
    void Execute(); // 실행
    void Undo(); // 취소 (되돌리기)
}
public class MoveCommand : ICommand
{
    private Unit receiver;
    private Vector2 targetPos; // 현재 상태
    private Vector2 prevPos; // 과거 상태 저장용
    public MoveCommand(Unit receiver, Vector2 targetPos)
    {
        this.receiver = receiver;
        this.targetPos = targetPos;
    }
    public void Execute()
    {
        prevPos = receiver.position;
        receiver.MoveTo(targetPos);
    }
    public void Undo()
    {
        receiver.MoveTo(prevPos);
    }
}
public class BuildCommand : ICommand
{
    private Unit receiver;
    private string buildingName;
    private Vector2 builtPos;
    public BuildCommand(Unit receiver, string buildingName)
    {
        this.receiver = receiver;
        this.buildingName = buildingName;
    }
    public void Execute()
    {
        builtPos = receiver.position;
        receiver.Build(builtPos, buildingName);
    }
    public void Undo()
    {
        receiver.RemoveBuilding(builtPos, buildingName);
    }
}

public class Unit
{
    public Vector2 position = new();

    // 명령어 목록
    public void MoveTo(Vector2 position)
    {
        this.position = position;
        Debug.Log($"(X:{position.x},Y:{position.y})로 움직임!");
    }
    public void Build(Vector2 pos, string name)
    {
        Debug.Log($"[Unit] {pos} 위치에 '{name}'을(를) 건설했습니다.");
    }
    public void RemoveBuilding(Vector2 pos, string name)
    {
        Debug.Log($"[Unit] {pos} 위치의 '{name}'을(를) 삭제했습니다.");
    }
}

public class GameLogic : MonoBehaviour
{
    // 히스토리를 관리할 스택 (가장 최근 명령이 맨 위에 쌓임)
    private Stack<ICommand> history = new Stack<ICommand>();
    private Unit workman = new();

    void Update()
    {
        // Z 키를 누르면 오른쪽으로 1칸 이동
        if (Keyboard.current.zKey.wasPressedThisFrame)
        {
            Vector2 targetPos = workman.position + new Vector2(1, 0);
            ExecuteCommand(new MoveCommand(workman, targetPos));
        }

        // X 키를 누르면 현재 위치에 배럭 건설
        if (Keyboard.current.xKey.wasPressedThisFrame)
        {
            ExecuteCommand(new BuildCommand(workman, "배럭"));
        }

        // spacebar 키를 누르면 실행 취소(Undo)
        if (Keyboard.current.spaceKey.wasPressedThisFrame)
        {
            Debug.Log("--- 실행 취소(Undo) 발동! ---");
            UndoLastCommand();
        }
    }

    // 명령 실행 + 히스토리 저장
    public void ExecuteCommand(ICommand command)
    {
        command.Execute();
        history.Push(command); // 나중을 위해 스택에 명령서를 밀어 넣음
    }
    // 실행 취소 (Undo / Ctrl+Z)
    public void UndoLastCommand()
    {
        if (history.Count > 0)
        {
            ICommand lastCommand = history.Pop();
            lastCommand.Undo();
        }
        else
        {
            Debug.Log("더 이상 취소할 명령이 없습니다.");
        }
    }
}

      이제 커맨드 패턴을 통해 이전에 무엇을 했는지 함수를 객체로 저장 할 수 있게 되었습니다.

      스페이스바를 통해 이전에 했던 작업을 Undo를 통해 취소 할 수도 있습니다.

      만약에 Undo 한 것을 다시 Redo 해보고 싶다면, Redo 기능을 코드를 직접 작성하여 커맨드 패턴이 어떤 패턴인지 연습해도 좋습니다.

      적용 했을 때 실행 화면

      이제 스페이스바를 통해 실행됬던 함수를 다시 되돌리는 걸 확인 할 수 있습니다.

      장점과 단점

      장점

      • 단일 책임 원칙(SRP): 작업을 요청하는 클래스(Invoker)와 작업을 수행하는 클래스(Receiver)를 완벽하게 분리할 수 있음
      • 개방-폐쇄 원칙(OCP): 기존 코드 수정 없이 새로운 명령어 클래스를 쉽게 추가할 수 있음
      • 확장성: 실행 취소(Undo), 재실행(Redo), 매크로(여러 명령 조합) 기능을 구현하기가 매우 수월함
      • 지연 실행: 요청을 큐(Queue)에 넣고 필요할 때 순차적으로 혹은 비동기적으로 처리할 수 있음

      단점

      • 클래스 폭발: 각 명령어마다 별도의 클래스를 만들어야 하므로 관리해야 할 클래스의 개수가 급격하게 늘어남
      • 구조적 복잡성: 단순한 기능 호출이라면 굳이 커맨드 패턴을 적용하여 간접 참조 계층을 늘릴 필요가 없어 오버엔지니어링이 될 위험이 존재함

      전체 요약

      1. 행위(명령) 자체를 캡슐화하여 객체로 다룰 수 있게 해주는 패턴
      2. Undo/Redo 기능 구현이나 명령어 큐잉 등 유연하고 확장성 있는 제어가 필요할 때
      3. 단, 남용할 경우 불필요한 클래스가 많아지므로 상황에 맞게 적절히 적용해야 함
