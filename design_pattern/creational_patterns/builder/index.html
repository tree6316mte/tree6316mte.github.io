      빌더(Builder) 패턴이란


      빌더(Builder) 패턴은 객체를 생성할 때 복잡한 객체의 생성 과정과 표현 방법을 분리하여 다양한 구성의 인스턴스를 만들기 위한 패턴이다.

      쉽게 말해, 매개변수에 멤버변수에 대한 값을 전달 하지 않고, 함수를 호출하여 멤버변수를 수정하는 방식입니다.

      빌더 패턴 사용 예시

      빌더 패턴 적용이 안된 코드와 적용된 코드를 만들어 비교해 보겠습니다.

      빌더 패턴 적용 Before 

      using UnityEngine;

public class Hamburger
{
    // 필수 매개변수
    private int bun;

    // 선택 매개변수
    private int patty, cheese, lettuce, tomato, bacon;

    public Hamburger(int bun, int patty, int cheese, int lettuce, int tomato, int bacon)
    {
        this.bun = bun;
        this.patty = patty;
        this.cheese = cheese;
        this.lettuce = lettuce;
        this.tomato = tomato;
        this.bacon = bacon;
    }

    public Hamburger(int bun, int patty, int cheese)
    {
        this.bun = bun;
        this.patty = patty;
        this.cheese = cheese;
    }
}

public class GameLogic : MonoBehaviour
{
    void Start()
    {
        // 모든 재료가 있는 햄버거
        Hamburger allBurger = new Hamburger(2, 8, 6, 7, 6, 7);

        // 빵과 패티 치즈만 있는 햄버거
        Hamburger cheeseBurger = new Hamburger(2, 1, 1);

        // 고기가 안 들어간 햄버거
        Hamburger veganBurger = new Hamburger(2, 0, 0, 6, 2, 0);
    }
}

      위 코드처럼 빌더 패턴을 적용하지 않고 다양한 버거를 만들 때는 생성자에 매개변수를 오버로딩해서 여러개 만들어야해서 작업이 번거로워집니다.

      빌더 패턴 적용 After 

      using UnityEngine;

public class Hamburger
{
    // 필수 매개변수
    private int bun;

    // 선택 매개변수
    private int patty, cheese, lettuce, tomato, bacon;

    private Hamburger() { }

    public class Builder
    {
        private Hamburger burger = new Hamburger();

        public Builder SetBun(int bun) { burger.bun = bun; return this; }
        public Builder SetPatty(int patty) { burger.patty = patty; return this; }
        public Builder SetCheese(int cheese) { burger.cheese = cheese; return this; }
        public Builder SetLettuce(int lettuce) { burger.lettuce = lettuce; return this; }
        public Builder SetTomato(int tomato) { burger.tomato = tomato; return this; }
        public Builder SetBacon(int bacon) { burger.bacon = bacon; return this; }

        public Hamburger Build()
        {
            // 필수 재료 검증
            if (burger.bun == 0) Debug.Log("빵 필수!");

            return burger;
        }
    }
}

public class GameLogic : MonoBehaviour
{
    void Start()
    {
        // 모든 재료가 있는 햄버거
        Hamburger allBurger = new Hamburger.Builder()
        .SetBun(2)
        .SetPatty(8)
        .SetCheese(6)
        .SetLettuce(7)
        .SetTomato(6)
        .SetBacon(7)
        .Build();
        
        // 빵과 패티 치즈만 있는 햄버거
        Hamburger cheeseBurger = new Hamburger.Builder()
        .SetBun(2)
        .SetPatty(1)
        .SetCheese(1)
        .Build();

        // 고기가 안 들어간 햄버거
        Hamburger veganBurger = new Hamburger.Builder()
        .SetBun(2)
        .SetLettuce(6)
        .SetTomato(2)
        .Build();
    }
}

      코드를 보면 생성자 대신 Builder라는 클래스를 하나 생성해서 설정할 멤버변수에 대해 Set함수를 만들고 마지막에 Build()함수를 통해 햄버거를 넘겨주게 됩니다.

      빌더 패턴을 사용하는 이유

      만약 빌더 패턴을 사용하지 않는다면?

      1. 멤버변수가 늘어나거나 조금 다른 객체를 생성할 때 생성자를 늘려야함

      사용한다면?

      1. 여러개의 조금씩 다른 객체에 대한 생성에 유연해짐

      전체 요약

      1. 인스턴스를 원하는 것에 맞게 설정 후 생성할 때 편리
      2. 매개변수가 많아져도 생성자를 여러 개 만들 필요 없어서 확장에 유연
